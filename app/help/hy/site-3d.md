3D դիտումը ցույց է տալիս տեղանքը որպես մեկ մոդել. գոյություն ունեցող գետինը և ձևավորված մակերևույթը, դրանց վրա դրված հորիզոնականները, կոտրման գծերը և մակերևույթի եզրը, յուրաքանչյուր երթուղի իր նախագծային պրոֆիլի վրա՝ պիկետների նշիչներով, միջանցքի գծային տարրերը և ինքը ճանապարհը (գոտիներ, ուսուկներ, ջրատարներ, եզրաքարեր, շերտեր և մայթեր՝ իրենց գույներով), հարթակները իրենց հատման գծերով, խողովակների ցանցը՝ խողովակներով և կառույցներով իրենց իրական բարձրությամբ, և Landscape աշխատանքային տարածքից՝ բույսերը, սալահատակը, արահետները, պատերը, եզրաքարերը, ռամպերն ու աստիճանները։ Այն կարդում է նույն նախագիծը, ինչ հատակագիծը, այնպես որ յուրաքանչյուր փոփոխություն երևում է երկուսում։

## Բացել 3D դիտումը

1. Բացեք հանույթի կետերը (կամ LandXML կամ նախագծի ֆայլը), որ լինի մակերևույթ։ Բույսերը, սալահատակը կամ ռամպերը բավական են դա բացելու համար։
2. Սեղմեք {{key:D}}, կամ սեղմեք {{ui:3D}}-ը {{ui:View}} խմբում՝ գործիքագոտում։
   3D դիտումը բացվում է հատակագծի վրայով՝ նայելով ամբողջ տեղանքին հարավ-արևելքից։ Առաջին անգամ այն մի պահ է պահանջում բեռնելու համար։
3. Սեղմեք {{key:D}}-ն նորից, կամ {{ui:3D}}-ն նորից, վերադառնալու համար հատակագծին։ Գծման գործիքի ընտրությունը, ինչպես {{ui:Alignment}} կամ {{ui:Pad}}, նույնպես վերադարձնում է հատակագիծ, քանի որ գծումը տեղի է ունենում այնտեղ։

## Տեղաշարժվել

| Ինչ | Արա սա |
|---|---|
| Պտտել | Քաշեք ձախ կամ միջին կոճակով |
| Տեղաշարժել | Քաշեք աջ կոճակով, կամ {{key:Shift}} և քաշեք |
| Մեծացնել | Պտտեք անիվը. մեծանում է դեպի ցուցիչը |
| Պտտել թաչպադով | Սահեցրեք երկու մատով ({{key:Shift}} և սահեցնել՝ տեղաշարժում) |
| Մեծացնել թաչպադով | Սեղմեք (pinch) |
| Հպում | Մեկ մատը պտտում է. երկու մատը սեղմում է մեծացնելու և տեղաշարժելու համար |
| Ստեղներ (դիտումն ունի ֆոկուս) | {{key:Left}} {{key:Right}} {{key:Up}} {{key:Down}} պտտում են, {{key:Shift}}-ով՝ տեղաշարժում. {{key:Plus}} և {{key:Minus}}՝ մեծացնում |

Դիտման վերևի գոտին ունի մնացածը.

- Դիտման ցանկ՝ **Իզո**, **Վերև**, **Ներքև**, **Առաջ** (հյուսիսին նայելով), **Հետև**, **Ձախ**, **Աջ**։
- {{ui:Perspective}}. սեղմված է՝ պերսպեկտիվ դիտում է. սեղմեք այն զուգահեռ պրոյեկցիայի համար (առանց կրճատման)։
- {{ui:Extents}}. ցույց է տալիս ամեն ինչ նորից։ {{key:F}}-ը նույնն է անում, երբ 3D դիտումը բաց է։

## Դիտման խորանարդ

Վերևում աջ անկյունում դիտման խորանարդը պտտվում է դիտման հետ։ Սեղմեք կողմ, եզր կամ անկյուն՝ տեղանքին այնտեղից նայելու համար (տեսախցիկը հարթ պտտվում է, եթե համակարգը չի խնդրում նվազեցված շարժում)։ Քաշեք խորանարդը՝ շրջանաձև պտտելու համար։ Դրա տակ գտնվող կողմնացույցի օղակը ցույց է տալիս հյուսիսը (ցանցի հյուսիսը, +Y). քաշեք օղակը՝ տեղանքը ուղղահայացի շուրջ պտտելու համար, կամ սեղմեք N, E, S կամ W՝ այդ կողմից նայելու համար։ {{ui:Home}}-ը, փոքր տունը նրա վերևի ձախ մասում, վերադարձնում է ամբողջ տեղանքի իզո դիտմանը։ Ստեղնաշարով՝ {{key:Tab}}՝ խորանարդ գնալու համար, սլաքները ընտրում են կողմ, եզր կամ անկյուն, {{key:Enter}}՝ այնտեղից նայելու համար։

Հատակագծի վրա նույն խորանարդը ցույց է տալիս վերևը. սեղմեք ցանկացած այլ կողմ, եզր կամ անկյուն (կամ քաշեք այն)՝ 3D դիտումը բացելու համար այնտեղից նայելով. սա ամենաարագ ճանապարհն է 3D մտնելու որևէ ուղղությունից։ Խորանարդը փոքր առանցքի նշանով փոխարինելու կամ թաքցնելու համար տես **Անկյունային ցուցիչ** [CivilCraft-ի կարգավորումներում](topic:settings)։

## Ուղղահայաց չափազանցություն

Տեղանքները սովորաբար շատ ավելի լայն են, քան բարձր, այնպես որ բարձրությունները սկզբում ձգվում են **2×**-ով։ Ընտրեք **1×**, **2×**, **3×**, **5×** կամ **10×** գոտու ցանկում կամ [CivilCraft-ի կարգավորումներում](topic:settings) **3D ուղղահայաց չափազանցություն**-ի տակ (երկուսը մեկ կարգավորում են)։ Փոխվում է միայն դիտումը. ծավալները, պրոֆիլները և արտահանումները մնում են իրական մասշտաբով։

## Գունավորել մակերևույթները

- **Բարձրության գոտիներ**. նույն գոտիներն ու գույները, ինչ հատակագծի բարձրության գոտիները (գոտիների քանակը սահմանվում է **Ցուցադրում**-ի տակ ձախ վահանակում)։ Յուրաքանչյուր գոտու եզրը ճշգրիտ կտրվում է իր բարձրության վրա։
- **Մեկ գույն**. գոյություն ունեցող գետինը և ձևավորված մակերևույթը՝ յուրաքանչյուրը պարզ գույնով։
- **Քարտեզային շերտեր**. հատակագծի տակ ցուցադրված քարտեզային շերտերը (գեոհղված նկար, քարտեզի սալիկներ կամ քարտեզային ծառայություն. տես [Քարտեզային շերտեր](topic:map-layers)) փռված մակերևույթների վրա որպես թերթ, այնպես որ ավիա լուսանկարը երևում է բլուրների վրա։ Նկարները, որոնք դեռ գալիս են, լցվում են, երբ գալիս են։ Երբ ոչ մի քարտեզային շերտ չի ցուցադրվում, մակերևույթները պահում են իրենց բարձրության գոտիները, և դիտումն ասում է դա։ Նկարը էկրանի համար է. GLB և OBJ արտահանումները այն չեն պարունակում։

Երբ երկու մակերևույթն էլ ցուցադրվում է, գոյություն ունեցող գետինը թափանցիկ է ձևավորված մակերևույթի վրա, այնպես որ կարող եք տեսնել հանումը (գետինը նախագծից վերև) և լցումը։

| Մակերևույթի եզր | Յուրաքանչյուր ցուցադրված մակերևույթի արտաքին եզրը, իր նիշերով |
| Կոտրման գծեր | Յուրաքանչյուր կոտրման գիծ, իր նիշերով |

| Բույսեր | Յուրաքանչյուր բույս, և բույսերը մարգերում ու ցանկապատներում, որպես պարզ ձևեր՝ բույսերի ցանկից չափված. ծառը բունով է և կլոր պսակով (կոնաձև՝ սոճու համար, փունջ բարձր բնի վրա՝ արմավենու համար), թփերը, ցանկապատի բույսերը, բազմամյաները և գետնածածկերը գմբեթներ են, խոտերը՝ փնջեր, բարձրացողները՝ բարակ սյուներ։ Նրանք կանգնում են ձևավորված մակերևույթի (կամ գոյություն ունեցող գետնի) վրա այն չափով, ինչ ցույց է տալիս հատակագիծը. հասուն կամ տնկման պահին |

| Խողովակներ և կառույցներ | Խողովակները՝ որպես կլոր խողովակներ իրենց ներքին հատակների միջև, կառույցները՝ եզրից մինչև ցածր հատված |
| Հանույթի կետեր | Կետ յուրաքանչյուր կետում՝ բարձրությամբ (սկզբում անջատված) |

Պիտակները մնում են նույն չափով և ուղիղ, ինչպես էլ շրջեք կամ մեծացնեք, և բլուրը պիտակի առջև թաքցնում է այն։ Երբ դրանք խտանում են, ամենակարևորը ցուցադրվում է, իսկ մնացածը սպասում է, մինչև մեծացնեք. նախ երթուղիները, ապա կառույցները, հարթակները, գծային տարրերը, պիկետները, հորիզոնականները և կետերը, ավելի մոտ եղածները առաջինը։ Դրանք միայն էկրանին են. արտահանումներն ու գծագրերը դրանք թողնում են։

Ճանապարհը կառուցվում է գոյություն ունեցող գետնի վրա։ Երթուղի և միջանցք, բայց առանց հանույթի կետերի կամ մակերևույթի, ճանապարհը նկարվում է հարթ փոխարինող գետնի վրա՝ առայժմ. իր նախագծային պրոֆիլի միջին բարձրության վրա, կամ մակարդակով 0 բարձրության վրա, երբ պրոֆիլ էլ չկա։ 3D դիտման ներքևում և միջանցքի հատկություններում զգուշացում է դա ասում։ Դա միայն նկար է. ոչ ծավալներ, ոչ կտրվածքներ չեն գալիս դրանից։ Բացեք հանույթի կետերը կամ մակերևույթը, և ճանապարհը ինքն իրեն տեղափոխվում է իրական գետնի վրա։

Ձեր ընտրությունները, ուղղահայաց չափազանցությունը և գունավորումը այս դիտարկիչում հիշվում են։

## Խողովակներ և կառույցներ

- Խողովակը նկարվում է իր ներքևի հատակը ներսում՝ դրած ներքին մակարդակներում, և նրա վերևը (գագաթը) մեկ ներքին տրամագիծ ավելի բարձր։ Տրամագծերը դյույմներով են ֆուտային նախագծում և միլիմետրերով մետրային նախագծում, ինչպես աջ վահանակում։
- Կառույցը գնում է իր եզրից (մուտքագրված կամ մակերևույթից) ներքև՝ մինչև իր ցածր հատվածը (ամենացածր ներքին մակարդակը մինուս ցածր հատվածի խորությունը)։ Առանց որևէ ներքին մակարդակի այն նկարվում է 5 ֆտ (1.5 մ) խորությամբ։
- Խնդիր ունեցող խողովակները և կառույցները (տես ցանցի ստուգումը աջ վահանակում) նկարվում են կարմիրով։
- Խողովակը առանց ներքին մակարդակների չի նկարվում. դիտման ներքևում նշումն ասում է, թե քանիսն են։

## Ընտրել 3D դիտումից

Սեղմեք երթուղին (կամ նրա ճանապարհը), հարթակը, գծային տարրը, խողովակը, կառույցը, հանույթի կետը, բույսը, ռամպը կամ այլ կոշտ ծածկույթ։ Այն ընտրվում է այնպես, կարծես սեղմել եք հատակագծի վրա. իր հատկությունները բացվում են աջ վահանակում, որտեղ կարող եք խմբագրել դրանք, և այն նկարվում է ընտրության գույնով՝ 3D-ում ամեն ինչից վերև։ Այն, ինչ ընտրում եք հատակագծի վրա, 3D-ում նույնպես առանձնացվում է։ Ցուցիչի կողքին գտնվող հուշումը անվանում է այն, ինչ սեղմել եք՝ իր արևելյան, հյուսիսային և բարձրության կոորդինատներով։ Խողովակներն ու կառույցները կարելի է սեղմել նույնիսկ այնտեղ, որտեղ նրանք գետնի տակ են. բլուրի հետևում թաքնված գիծը չի կարելի սեղմել դրա միջով։

Սեղմեք կոտրման գիծը՝ տեսնելու, թե որն է. այն առանձնանում է, և կարգավիճակի գոտին անվանում է այն (կոտրման գծերն իրենց հատկություններ չունեն)։

Սեղմեք գետինը՝ հուշման մեջ կարդալու արևելյան, հյուսիսային և բարձրության կոորդինատները այնտեղ։

## Վարել երթուղիով

Սեղմեք **Վարել** 3D գոտում և ընտրեք երթուղի։ Դիտումը տեղափոխվում է վարորդի նստատեղը՝ 3.5 ֆտ (1.07 մ) ճանապարհից վեր, իր նախագծային պրոֆիլի վրա (գոյություն ունեցող գետնի վրա կենտրոնագծով, երբ դեռ չունի), նայելով առաջ դեպի այն։

- **Նվագարկել** կամ {{key:Space}}-ը վարում է ձեր ընտրած արագությամբ (10-ից 65 մղոն/ժամ, կամ 15-ից 100 կմ/ժ). {{key:Space}}-ը նորից դադարեցնում է։ Վերջում այն կանգ է առնում. **Նվագարկել**-ը այն ժամանակ նորից սկսում է սկզբից։
- Սլայդերը անցնում է ցանկացած պիկետի, {{key:Left}} և {{key:Right}}-ը քայլում են 10 ֆտ (5 մ) հետ և առաջ։
- Քաշելը դիտման մեջ դադարեցնում է և թույլ է տալիս նայել շուրջը։
- **Կանգնեցնել** կամ {{key:Esc}}-ը դիտումը վերադարձնում է այնտեղ, որտեղ եղել է։

Ուղղահայաց չափազանցությունը կիրառվում է նաև վարելու ընթացքում. սահմանեք այն 1×, որ տեսնեք ճանապարհը այնպես, ինչպես այն կլինի։

## Արևի լույսը

Բացեք **Արև** 3D գոտում և նշեք **Լույս արևից**. տվեք ամսաթիվ, ժամացույցի ժամ և դրա ժամային գոտին (շեղումը UTC-ից. ամռանը՝ ամառային ժամանակի)։ Մուտքագրեք նաև տեղանքի աշխարհագրական լայնությունը և երկայնությունը։ Երբ նախագիծն ունի կոորդինատային համակարգ, դրանք լրացվում են դրանից։

Տեղանքը այնուհետև լուսավորվում է այնտեղից, որտեղ գտնվում է արևը, **Ստվերներով**՝ գետնի, միջանցքի և հարթակների վրա. ցանկը ասում է արևի ազիմուտը և ինչքան բարձր է, կամ որ այն հորիզոնից ցածր է։ Դիրքերը գալիս են NOAA-ի արևային հավասարումներից։ Ստվերները ճիշտ ուղղությամբ են ընկնում ցանկացած չափազանցությամբ, բայց կանգնում են տեղանքի եզրին. այդ եզրից դուրս գտնվող բլուրներն ու շենքերը ստվեր չեն ձգում։

## Արտահանել 3D դիտումը

Սեղմեք {{ui:GLB}} կամ {{ui:OBJ}} գոտում։ Այն, ինչ ցույց է տալիս դիտումը, պահվում է իրական մասշտաբով (առանց չափազանցության), մետրերով, Y-ը վերև։ Մանրամասներ՝ [Արտահանել տեղանքը 3D-ում](topic:export-3d)։
